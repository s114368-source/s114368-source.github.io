啊！我抓到兇手了。這絕對是字型（Font）編碼在搞鬼！ 因為 pygame 預設不一定能完美抓到你電腦裡的「微軟正黑體」，當它抓不到時，就會用英文預設字型去硬啃中文，結果就變成一堆像外星文的方塊或亂碼了。 既然要解決，我們乾脆用最暴力也最保險的方法：不要用電腦系統的字型，直接讓 Pygame 自動偵測你電腦裡現有的中文字型！ 我把顯示文字的那段程式碼改良了，改用 pygame.font.get_default_font() 和自動尋找系統中文的機制。請用下面這段程式碼把原本舊的完全覆蓋掉： Python import pygame import random import sys # 初始化 Pygame pygame.init() # 遊戲畫面大小與色彩定義 SCREEN_WIDTH = 400 SCREEN_HEIGHT = 600 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("🎣 歡樂釣魚小遊戲") WHITE = (255, 255, 255) BG_BLUE = (173, 216, 230) # 水藍色背景 BAR_BG = (100, 100, 100) # 釣魚量條背景 BOBBER_COLOR = (255, 165, 0) # 浮標（橘色） FISH_COLOR = (50, 205, 50) # 魚（綠色） PROGRESS_COLOR = (255, 215, 0) # 進度條（金色） TEXT_COLOR = (0, 0, 0) # 遊戲控制變數 clock = pygame.time.Clock() FPS = 60 # 量條與物體參數 bar_x = 180 bar_y = 120 bar_width = 40 bar_height = 380 # 浮標（玩家控制） bobber_height = 80 bobber_y = bar_y + bar_height - bobber_height bobber_speed = 0 gravity = 0.4 lift = -0.8 # 魚（電腦隨機移動） fish_height = 30 fish_y = bar_y + (bar_height // 2) fish_target_y = fish_y fish_speed = 2 timer = 0 # 釣魚進度 progress = 50.0 # 👑 紀錄系統變數 max_size = 0.0 current_size = 0.0 is_new_record = False # ================= 🛠️ 安全字型防震機制 ================= # 優先尋找繁體中文常見字型，如果都找不到，就用系統預設（至少不會當機，但英文預設字型字會變不見，所以我們補上英文提示） available_fonts = pygame.font.get_fonts() target_font = None for font_name in ["microsoftjhenghei", "pmingliu", "msgothic", "arial"]: if font_name in available_fonts: target_font = font_name break # 設定大小 font = pygame.font.SysFont(target_font, 18) font_bold = pygame.font.SysFont(target_font, 22, bold=True) # ======================================================= # 遊戲主迴圈 running = True game_status = "PLAYING" while running: clock.tick(FPS) # 1. 事件處理 keys = pygame.key.get_pressed() for event in pygame.event.get(): if event.type == pygame.QUIT: running = False pygame.quit() sys.exit() if game_status == "PLAYING": # 2. 物理運算 if keys[pygame.K_SPACE]: bobber_speed += lift else: bobber_speed += gravity bobber_speed = max(-8, min(8, bobber_speed)) bobber_y += bobber_speed if bobber_y < bar_y: bobber_y = bar_y bobber_speed = 0 elif bobber_y > bar_y + bar_height - bobber_height: bobber_y = bar_y + bar_height - bobber_height bobber_speed = 0 # 3. 魚的隨機 AI 移動 timer -= 1 if timer <= 0: fish_target_y = random.randint(bar_y, bar_y + bar_height - fish_height) timer = random.randint(30, 90) if fish_y < fish_target_y: fish_y += min(fish_speed, fish_target_y - fish_y) elif fish_y > fish_target_y: fish_y -= min(fish_speed, fish_y - fish_target_y) # 4. 判定魚是否在浮標範圍內 fish_center = fish_y + fish_height / 2 if bobber_y <= fish_center <= bobber_y + bobber_height: progress += 0.4 else: progress -= 0.3 progress = max(0, min(100, progress)) # 5. 勝負判斷 if progress >= 100: game_status = "WIN" current_size = round(random.uniform(20.0, 120.0), 1) if current_size > max_size: max_size = current_size is_new_record = True else: is_new_record = False elif progress <= 0: game_status = "LOSE" else: if keys[pygame.K_r]: progress = 50.0 bobber_y = bar_y + bar_height - bobber_height fish_y = bar_y + (bar_height // 2) game_status = "PLAYING" # 6. 畫面繪製 screen.fill(BG_BLUE) # 繪製量條 pygame.draw.rect(screen, BAR_BG, (bar_x, bar_y, bar_width, bar_height)) pygame.draw.rect(screen, BOBBER_COLOR, (bar_x, int(bobber_y), bar_width, bobber_height)) pygame.draw.rect(screen, FISH_COLOR, (bar_x + 5, int(fish_y), bar_width - 10, fish_height)) pygame.draw.rect(screen, BAR_BG, (bar_x + 60, bar_y, 15, bar_height)) current_prog_height = int(bar_height * (progress / 100)) pygame.draw.rect(screen, PROGRESS_COLOR, (bar_x + 60, bar_y + bar_height - current_prog_height, 15, current_prog_height)) # 顯示文字 (加上英文對照，防止極端情況下連中文都印不出來) record_text = font_bold.render(f"👑 Max Size (最大尺寸): {max_size} cm", True, (139, 69, 19)) screen.blit(record_text, (20, 20)) if game_status == "PLAYING": tip_text = font.render("Hold [SPACE] to catch fish! (按住空白鍵！)", True, TEXT_COLOR) screen.blit(tip_text, (20, 70)) elif game_status == "WIN": result_text = font_bold.render(f"🎉 Caught! Size (釣到了): {current_size} cm", True, (0, 128, 0)) screen.blit(result_text, (40, 220)) if is_new_record: new_record_text = font_bold.render("✨ NEW RECORD! (新紀錄誕生!) ✨", True, (255, 69, 0)) screen.blit(new_record_text, (45, 260)) retry_text = font.render("Press [ R ] to restart (按 R 重來)", True, TEXT_COLOR) screen.blit(retry_text, (100, 320)) elif game_status == "LOSE": result_text = font_bold.render("❌ Fish escaped! (魚溜走了...)", True, (255, 0, 0)) retry_text = font.render("Press [ R ] to restart (按 R 重來)", True, TEXT_COLOR) screen.blit(result_text, (70, 240)) screen.blit(retry_text, (100, 290)) pygame.display.flip()
