import pygame import random import sys # 初始化 Pygame pygame.init() # 遊戲畫面大小與色彩定義 SCREEN_WIDTH = 400 SCREEN_HEIGHT = 600 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("🎣 歡樂釣魚小遊戲") WHITE = (255, 255, 255) BG_BLUE = (173, 216, 230) # 水藍色背景 BAR_BG = (100, 100, 100) # 釣魚量條背景 BOBBER_COLOR = (255, 165, 0) # 浮標（橘色） FISH_COLOR = (50, 205, 50) # 魚（綠色） PROGRESS_COLOR = (255, 215, 0) # 進度條（金色） TEXT_COLOR = (0, 0, 0) # 遊戲控制變數 clock = pygame.time.Clock() FPS = 60 # 量條與物體參數 bar_x = 180 bar_y = 120 # 稍微往下挪一點，留空間給上方的紀錄看板 bar_width = 40 bar_height = 380 # 浮標（玩家控制） bobber_height = 80 bobber_y = bar_y + bar_height - bobber_height bobber_speed = 0 gravity = 0.4 lift = -0.8 # 魚（電腦隨機移動） fish_height = 30 fish_y = bar_y + (bar_height // 2) fish_target_y = fish_y fish_speed = 2 timer = 0 # 釣魚進度 progress = 50.0 # 滿分 100 算釣到 # 👑 紀錄系統變數 max_size = 0.0 # 歷史最大尺寸 (cm) current_size = 0.0 # 本次釣到的尺寸 (cm) is_new_record = False # 是否破紀錄的旗標 # 遊戲主迴圈 running = True game_status = "PLAYING" # "PLAYING", "WIN", "LOSE" while running: clock.tick(FPS) # 1. 事件處理 keys = pygame.key.get_pressed() for event in pygame.event.get(): if event.type == pygame.QUIT: running = False pygame.quit() sys.exit() if game_status == "PLAYING": # 2. 物理運算：浮標控制 if keys[pygame.K_SPACE]: bobber_speed += lift else: bobber_speed += gravity bobber_speed = max(-8, min(8, bobber_speed)) bobber_y += bobber_speed # 限制浮標不超出量條 if bobber_y < bar_y: bobber_y = bar_y bobber_speed = 0 elif bobber_y > bar_y + bar_height - bobber_height: bobber_y = bar_y + bar_height - bobber_height bobber_speed = 0 # 3. 魚的隨機 AI 移動 timer -= 1 if timer <= 0: fish_target_y = random.randint(bar_y, bar_y + bar_height - fish_height) timer = random.randint(30, 90) if fish_y < fish_target_y: fish_y += min(fish_speed, fish_target_y - fish_y) elif fish_y > fish_target_y: fish_y -= min(fish_speed, fish_y - fish_target_y) # 4. 判定魚是否在浮標範圍內 fish_center = fish_y + fish_height / 2 if bobber_y <= fish_center <= bobber_y + bobber_height: progress += 0.4 else: progress -= 0.3 progress = max(0, min(100, progress)) # 5. 勝負判斷 if progress >= 100: game_status = "WIN" # 隨機決定這次釣到的魚尺寸（例如 20.0 到 120.0 公分，四捨五入到小數第一位） current_size = round(random.uniform(20.0, 120.0), 1) # 檢查是否打破最大尺寸紀錄 if current_size > max_size: max_size = current_size is_new_record = True else: is_new_record = False elif progress <= 0: game_status = "LOSE" else: # 如果輸或贏了，按 R 鍵可以重來 if keys[pygame.K_r]: progress = 50.0 bobber_y = bar_y + bar_height - bobber_height fish_y = bar_y + (bar_height // 2) game_status = "PLAYING" # 6. 畫面繪製 screen.fill(BG_BLUE) # 繪製主釣魚量條與進度條 pygame.draw.rect(screen, BAR_BG, (bar_x, bar_y, bar_width, bar_height)) pygame.draw.rect(screen, BOBBER_COLOR, (bar_x, int(bobber_y), bar_width, bobber_height)) pygame.draw.rect(screen, FISH_COLOR, (bar_x + 5, int(fish_y), bar_width - 10, fish_height)) pygame.draw.rect(screen, BAR_BG, (bar_x + 60, bar_y, 15, bar_height)) current_prog_height = int(bar_height * (progress / 100)) pygame.draw.rect(screen, PROGRESS_COLOR, (bar_x + 60, bar_y + bar_height - current_prog_height, 15, current_prog_height)) # 顯示文字提示與紀錄 font = pygame.font.SysFont("microsoftjhenghei", 22) font_bold = pygame.font.SysFont("microsoftjhenghei", 26, bold=True) # 頂部常駐的最高紀錄看板 record_text = font_bold.render(f"👑 歷史最大尺寸: {max_size} cm", True, (139, 69, 19)) screen.blit(record_text, (20, 20)) if game_status == "PLAYING": tip_text = font.render("按住 [空白鍵] 控制橘色浮標！", True, TEXT_COLOR) screen.blit(tip_text, (20, 70)) elif game_status == "WIN": result_text = font_bold.render(f"🎉 釣到了！尺寸：{current_size} cm", True, (0, 128, 0)) screen.blit(result_text, (50, 220)) if is_new_record: new_record_text = font_bold.render("✨ 新紀錄誕生！太強了！ ✨", True, (255, 69, 0)) screen.blit(new_record_text, (45, 260)) retry_text = font.render("按下 [ R ] 重新挑戰", True, TEXT_COLOR) screen.blit(retry_text, (110, 320)) elif game_status == "LOSE": result_text = font_bold.render("❌ 魚溜走了...再接再厲！", True, (255, 0, 0)) retry_text = font.render("按下 [ R ] 重新挑戰", True, TEXT_COLOR) screen.blit(result_text, (80, 240)) screen.blit(retry_text, (110, 290)) pygame.display.flip()
